package generated.uibuilder

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun ClickableStateLayout() {
    val page = androidx.compose.runtime.remember { androidx.compose.runtime.mutableStateOf<kotlin.Int>(10) }
    Box(modifier = Modifier.fillMaxSize().padding(24.dp)) {
        when (page.value) {
            10 -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(0xFF6750A4)).clickable(onClick = { page.value = 20 }), content = {})
            }
            20 -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(0xFF008577)).clickable(onClick = { page.value = 30 }), content = {})
            }
            else -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(0xFF3949AB)).clickable(onClick = { page.value = 10 }), content = {})
            }
        }
    }
}
