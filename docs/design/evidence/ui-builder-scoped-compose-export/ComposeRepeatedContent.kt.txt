package proof.repetition.compose

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.layout.size
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun ComposeRepeatedContent() {
    val page = androidx.compose.runtime.remember { androidx.compose.runtime.mutableStateOf<kotlin.Int>(10) }
    Column(content = {
        Column(verticalArrangement = Arrangement.spacedBy(4.dp), content = {
            data class ScreenRow(val field0: kotlin.Float)
            kotlin.collections.listOf<ScreenRow>(ScreenRow(0.0f), ScreenRow(8.0f), ScreenRow(16.0f)).forEach { screenRow ->
                Pair(modifier = Modifier.padding(start = 4.dp, top = 2.dp, end = 4.dp, bottom = 2.dp), argument0 = screenRow.field0, capture0 = { page.value = 10 }, capture1 = { page.value = 20 })
            }
        })
        Box(modifier = Modifier.size(width = 60.dp, height = 20.dp), content = {
            when (page.value) {
                10 -> {
                    Box(modifier = Modifier.fillMaxSize().background(color = Color(4294901760L)), content = {})
                }
                20 -> {
                    Box(modifier = Modifier.fillMaxSize().background(color = Color(4278255360L)), content = {})
                }
                else -> {
                    Box(modifier = Modifier.fillMaxSize().background(color = Color(4278190335L)), content = {})
                }
            }
        })
    })
}

@Composable
private fun Pair(modifier: androidx.compose.ui.Modifier, argument0: kotlin.Float, capture0: () -> kotlin.Unit, capture1: () -> kotlin.Unit) {
    Box(modifier = modifier, content = {
        Row(horizontalArrangement = Arrangement.spacedBy(argument0.dp), content = {
            Box(modifier = Modifier.size(width = 16.dp, height = 16.dp).background(color = Color(4294901760L)).clickable(onClick = capture0), content = {})
            Box(modifier = Modifier.size(width = 16.dp, height = 16.dp).background(color = Color(4278255360L)).clickable(onClick = capture1), content = {})
        })
    })
}
